cmake_minimum_required(VERSION 3.13...3.31)
if(NOT DEFINED PICO_SDK_PATH)
  if(DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
  else()
    message(FATAL_ERROR "PICO_SDK_PATH is not set. Use Pico SDK 2.3.0 or newer.")
  endif()
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(zoolark_rp2350 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
list(APPEND PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_LIST_DIR}/include/boards)
pico_sdk_init()
add_executable(zoolark_rp2350 src/main.c src/usb_descriptors.c src/wfl_protocol.c src/logic_capture.c src/jtag_bitbang.c src/kxtj3.c src/ws2812_status.c)
target_include_directories(zoolark_rp2350 PRIVATE ${CMAKE_CURRENT_LIST_DIR}/include)
pico_generate_pio_header(zoolark_rp2350 ${CMAKE_CURRENT_LIST_DIR}/src/logic_capture.pio)
pico_generate_pio_header(zoolark_rp2350 ${CMAKE_CURRENT_LIST_DIR}/src/ws2812_status.pio)
target_compile_definitions(zoolark_rp2350 PRIVATE PICO_STDIO_USB=0 PICO_STDIO_UART=0)
target_link_libraries(zoolark_rp2350 PRIVATE pico_stdlib pico_unique_id pico_bootrom hardware_adc hardware_i2c hardware_pio hardware_dma hardware_clocks hardware_gpio tinyusb_device)
pico_enable_stdio_usb(zoolark_rp2350 0)
pico_enable_stdio_uart(zoolark_rp2350 0)
pico_set_program_name(zoolark_rp2350 "ZooLark RP2350B Dual ADC Firmware")
pico_set_program_version(zoolark_rp2350 "0.4.4-dual-adc")
pico_add_extra_outputs(zoolark_rp2350)
